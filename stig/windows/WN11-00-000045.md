# STIG ID: [WN11-00-000045](https://stigaview.com/products/win11/v2r5/WN11-00-000045/)

## Vulnerability Discussion

Malicious software can establish a base on individual desktops and servers. Employing an automated mechanism to detect this type of software will aid in elimination of the software from the operating system.
Check

## Compliance Check
- Verify an antivirus solution is installed on the system and in use. The antivirus solution may be bundled with an approved Endpoint Security Solution.
- Verify if Microsoft Defender Antivirus is in use or enabled:
- Open "PowerShell".
- Enter `"get-service | where {$_.DisplayName -Like "*Defender*"} | Select Status,DisplayName"`
- Verify third-party antivirus is in use or enabled:
- Open "PowerShell".
- Enter `"get-service | where {$_.DisplayName -Like "*mcafee*"} | Select Status,DisplayName"`
- Enter `"get-service | where {$_.DisplayName -Like "*symantec*"} | Select Status,DisplayName"`
- If there is no antivirus solution installed on the system, this is a finding.

## Remediation
 Install Microsoft Defender Antivirus or a third-party antivirus solution.
 
## Powershell Remediation Script
```
<#
.SYNOPSIS
    STIG Remediation: WN11-00-000045
    The Windows 11 system must use an antivirus program.

.DESCRIPTION
    Severity:         CAT I (High)
    SRG:              SRG-OS-000480-GPOS-00227
    CCI:              CCI-000366
    Vulnerability ID: V-253264

    Checks for an active antivirus solution (Microsoft Defender or approved
    third-party AV). If Defender is present but disabled, this script will
    attempt to enable it.

.NOTES
    Run as Administrator.
    If a third-party AV is required by your org, install it manually.
#>

#Requires -RunAsAdministrator

# ─────────────────────────────────────────────
# LOGGING SETUP
# ─────────────────────────────────────────────
$LogPath = "C:\Logs\STIG"
$LogFile = "$LogPath\WN11-00-000045_$(Get-Date -Format 'yyyyMMdd_HHmmss').log"

if (-not (Test-Path $LogPath)) {
    New-Item -ItemType Directory -Path $LogPath -Force | Out-Null
}

function Write-Log {
    param (
        [string]$Message,
        [ValidateSet("INFO","WARN","ERROR","PASS","FAIL","FIX")]
        [string]$Level = "INFO"
    )
    $Timestamp = Get-Date -Format "yyyy-MM-dd HH:mm:ss"
    $Entry = "[$Timestamp][$Level] $Message"
    Write-Host $Entry -ForegroundColor $(switch ($Level) {
        "PASS"  { "Green"  }
        "FAIL"  { "Red"    }
        "WARN"  { "Yellow" }
        "FIX"   { "Cyan"   }
        "ERROR" { "Red"    }
        default { "White"  }
    })
    Add-Content -Path $LogFile -Value $Entry
}

# ─────────────────────────────────────────────
# BANNER
# ─────────────────────────────────────────────
Write-Log "=============================================" "INFO"
Write-Log " STIG ID   : WN11-00-000045"                  "INFO"
Write-Log " CAT       : I (High)"                         "INFO"
Write-Log " CCI       : CCI-000366"                       "INFO"
Write-Log " Vuln ID   : V-253264"                         "INFO"
Write-Log " Started   : $(Get-Date)"                      "INFO"
Write-Log "=============================================" "INFO"

# ─────────────────────────────────────────────
# STEP 1 — CHECK FOR THIRD-PARTY AV
# ─────────────────────────────────────────────
Write-Log "Checking for third-party antivirus solutions..." "INFO"

$ThirdPartyAV = @("mcafee", "symantec", "norton", "trend", "kaspersky",
                  "malwarebytes", "crowdstrike", "cylance", "sentinel",
                  "sophos", "eset", "bitdefender", "webroot", "avast", "avg")

$ThirdPartyFound = $false

foreach ($av in $ThirdPartyAV) {
    $svc = Get-Service | Where-Object { $_.DisplayName -like "*$av*" } -ErrorAction SilentlyContinue
    if ($svc) {
        foreach ($s in $svc) {
            Write-Log "Third-party AV detected: $($s.DisplayName) — Status: $($s.Status)" "INFO"
            if ($s.Status -eq "Running") {
                Write-Log "Third-party AV '$($s.DisplayName)' is RUNNING. Requirement satisfied." "PASS"
                $ThirdPartyFound = $true
            } else {
                Write-Log "Third-party AV '$($s.DisplayName)' is NOT running (Status: $($s.Status))." "WARN"
            }
        }
    }
}

if ($ThirdPartyFound) {
    Write-Log "Third-party AV is active. No further remediation needed." "PASS"
    Write-Log "Log saved to: $LogFile" "INFO"
    exit 0
}

Write-Log "No active third-party AV found. Checking Microsoft Defender..." "INFO"

# ─────────────────────────────────────────────
# STEP 2 — CHECK MICROSOFT DEFENDER SERVICES
# ─────────────────────────────────────────────
$DefenderServices = Get-Service | Where-Object { $_.DisplayName -like "*Defender*" }

if (-not $DefenderServices) {
    Write-Log "No Microsoft Defender services found on this system." "FAIL"
    Write-Log "MANUAL ACTION REQUIRED: Install an approved antivirus solution." "ERROR"
    exit 1
}

foreach ($svc in $DefenderServices) {
    Write-Log "Found: $($svc.DisplayName) — Status: $($svc.Status) | StartType: $($svc.StartType)" "INFO"
}

# ─────────────────────────────────────────────
# STEP 3 — CHECK DEFENDER ANTIVIRUS STATUS
# ─────────────────────────────────────────────
Write-Log "Querying Microsoft Defender Antivirus status via WMI..." "INFO"

try {
    $AVStatus = Get-MpComputerStatus -ErrorAction Stop

    Write-Log "AntivirusEnabled     : $($AVStatus.AntivirusEnabled)"     "INFO"
    Write-Log "RealTimeProtection   : $($AVStatus.RealTimeProtectionEnabled)" "INFO"
    Write-Log "AntispywareEnabled   : $($AVStatus.AntispywareEnabled)"   "INFO"
    Write-Log "AMServiceEnabled     : $($AVStatus.AMServiceEnabled)"     "INFO"
    Write-Log "AntivirusSignatureAge: $($AVStatus.AntivirusSignatureAge) day(s)" "INFO"

    if ($AVStatus.AntivirusEnabled -and $AVStatus.RealTimeProtectionEnabled) {
        Write-Log "Microsoft Defender Antivirus is ENABLED and Real-Time Protection is ON." "PASS"
        Write-Log "STIG WN11-00-000045: COMPLIANT" "PASS"
        Write-Log "Log saved to: $LogFile" "INFO"
        exit 0
    } else {
        Write-Log "Microsoft Defender is present but not fully enabled. Attempting remediation..." "WARN"
    }
}
catch {
    Write-Log "Could not query Get-MpComputerStatus: $_" "WARN"
    Write-Log "Defender may be managed by a third-party AV or GPO. Attempting service-level check..." "WARN"
}

# ─────────────────────────────────────────────
# STEP 4 — REMEDIATION: ENABLE DEFENDER
# ─────────────────────────────────────────────
Write-Log "Attempting to enable Microsoft Defender Antivirus..." "FIX"

# 4a — Ensure the WinDefend service is running
$WinDefend = Get-Service -Name "WinDefend" -ErrorAction SilentlyContinue
if ($WinDefend) {
    if ($WinDefend.StartType -ne "Automatic") {
        Write-Log "Setting WinDefend service StartType to Automatic..." "FIX"
        Set-Service -Name "WinDefend" -StartupType Automatic
    }
    if ($WinDefend.Status -ne "Running") {
        Write-Log "Starting WinDefend service..." "FIX"
        Start-Service -Name "WinDefend" -ErrorAction SilentlyContinue
        Start-Sleep -Seconds 5
        $WinDefend.Refresh()
        Write-Log "WinDefend service status after start attempt: $($WinDefend.Status)" "INFO"
    }
} else {
    Write-Log "WinDefend service not found. Defender may not be installed." "ERROR"
}

# 4b — Remove DisableAntiSpyware registry key if present (common tamper method)
$RegPath = "HKLM:\SOFTWARE\Policies\Microsoft\Windows Defender"
$RegName = "DisableAntiSpyware"

if (Test-Path $RegPath) {
    $RegValue = Get-ItemProperty -Path $RegPath -Name $RegName -ErrorAction SilentlyContinue
    if ($null -ne $RegValue) {
        Write-Log "Found '$RegName' registry key set to $($RegValue.$RegName). Removing..." "FIX"
        Remove-ItemProperty -Path $RegPath -Name $RegName -Force -ErrorAction SilentlyContinue
        Write-Log "'$RegName' registry key removed." "FIX"
    } else {
        Write-Log "'$RegName' registry key not present. No action needed." "INFO"
    }
}

# 4c — Enable Real-Time Protection via Defender preference
try {
    Write-Log "Setting Defender preferences: enabling Real-Time Monitoring..." "FIX"
    Set-MpPreference -DisableRealtimeMonitoring $false -ErrorAction Stop
    Write-Log "Real-Time Monitoring enabled successfully." "FIX"
}
catch {
    Write-Log "Could not set MpPreference: $_ — May be controlled by GPO." "WARN"
}

# ─────────────────────────────────────────────
# STEP 5 — FINAL VERIFICATION
# ─────────────────────────────────────────────
Write-Log "Running final compliance verification..." "INFO"

try {
    $AVStatusFinal = Get-MpComputerStatus -ErrorAction Stop

    if ($AVStatusFinal.AntivirusEnabled -and $AVStatusFinal.RealTimeProtectionEnabled) {
        Write-Log "FINAL STATUS: Microsoft Defender is ENABLED and Real-Time Protection is ON." "PASS"
        Write-Log "STIG WN11-00-000045: COMPLIANT" "PASS"
        exit 0
    } else {
        Write-Log "FINAL STATUS: Defender AV Enabled=$($AVStatusFinal.AntivirusEnabled) | RTP=$($AVStatusFinal.RealTimeProtectionEnabled)" "FAIL"
        Write-Log "STIG WN11-00-000045: NON-COMPLIANT — Manual intervention required." "FAIL"
        Write-Log "If a GPO is blocking Defender, coordinate with your administrator." "WARN"
        exit 1
    }
}
catch {
    Write-Log "Final verification query failed: $_" "ERROR"
    Write-Log "STIG WN11-00-000045: STATUS UNKNOWN — Manual review required." "WARN"
    exit 1
}
finally {
    Write-Log "Log saved to: $LogFile" "INFO"
    Write-Log "=============================================" "INFO"
}
```
